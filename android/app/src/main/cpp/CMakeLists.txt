cmake_minimum_required(VERSION 3.22.1)
project(clipforge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_AARCH64 ON CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/whisper whisper-build)
set(QJS ${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/quickjs)
add_library(quickjs-core STATIC ${QJS}/quickjs.c ${QJS}/cutils.c ${QJS}/dtoa.c ${QJS}/libregexp.c ${QJS}/libunicode.c)
target_include_directories(quickjs-core PUBLIC ${QJS})
target_compile_definitions(quickjs-core PRIVATE _GNU_SOURCE QUICKJS_NG_BUILD)
set_target_properties(quickjs-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(clipforge SHARED whisper_jni.cpp quickjs_jni.c)
target_link_libraries(clipforge whisper quickjs-core android log m)
target_link_options(clipforge PRIVATE "-Wl,-z,max-page-size=16384")
